cmake_minimum_required(VERSION 3.22)
project(SnapHost VERSION 0.0.1 LANGUAGES C CXX OBJCXX)
set(CMAKE_CXX_STANDARD 17)
set(JUCE_DIR "/tmp/juce-src" CACHE PATH "caminho do JUCE")
add_subdirectory(${JUCE_DIR} juce)
juce_add_gui_app(SnapHost PRODUCT_NAME "SnapHost")
target_sources(SnapHost PRIVATE main.cpp grab.mm)
target_compile_definitions(SnapHost PRIVATE JUCE_PLUGINHOST_VST3=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_APPLICATION_NAME_STRING="SnapHost" JUCE_APPLICATION_VERSION_STRING="0.1")
target_link_libraries(SnapHost PRIVATE juce::juce_audio_utils juce::juce_audio_processors juce::juce_recommended_config_flags)
target_link_libraries(SnapHost PRIVATE "-framework Cocoa")
